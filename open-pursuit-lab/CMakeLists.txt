cmake_minimum_required(VERSION 3.20)
project(OpenPursuitLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(opl_core src/core.cpp)
target_include_directories(opl_core PUBLIC include)
if(MSVC)
  target_compile_options(opl_core PRIVATE /W4 /WX)
else()
  target_compile_options(opl_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(opl_demo src/main.cpp)
target_link_libraries(opl_demo PRIVATE opl_core)
add_executable(opl_tests tests/core_tests.cpp)
target_link_libraries(opl_tests PRIVATE opl_core)
enable_testing()
add_test(NAME core_contracts COMMAND opl_tests)
add_test(NAME seeded_demo COMMAND opl_demo --seed 2026 --ticks 3600)
